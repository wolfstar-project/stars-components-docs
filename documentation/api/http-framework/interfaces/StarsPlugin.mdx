<ApiMember kind="property" badges={['optional']} signature={['enforce?: "pre" | "post"']} source={{ url: 'https://github.com/wolfstar-project/stars-components/blob/9d40129da57632753ba336f076c61dbe4a84131d/packages/http-framework/src/lib/plugins/definePlugin.ts#L42', label: 'packages/http-framework/src/lib/plugins/definePlugin.ts:42' }}>
